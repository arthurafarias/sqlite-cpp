# SRS 002 RR-5/SRS 001 FR-11/FR-12: mirrors every sqlite-cpp library's
# tests/CMakeLists.txt -- every testing/*_test.hpp header under this
# application's include/ tree self-registers a test_group on inclusion.
set(SQLITE_MKKEYWORDHASH_TEST_INCLUDE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../include")

set(GENERATED_TESTS "${CMAKE_CURRENT_BINARY_DIR}/all_tests.cpp")
file(WRITE "${GENERATED_TESTS}" "// Generated by CMake. Every testing/*_test.hpp header self-registers a test_group.\n")
file(GLOB_RECURSE SQLITE_MKKEYWORDHASH_TEST_HEADERS CONFIGURE_DEPENDS "${SQLITE_MKKEYWORDHASH_TEST_INCLUDE_ROOT}/*_test.hpp")
foreach(TEST_HEADER IN LISTS SQLITE_MKKEYWORDHASH_TEST_HEADERS)
    file(RELATIVE_PATH RELATIVE_HEADER "${SQLITE_MKKEYWORDHASH_TEST_INCLUDE_ROOT}" "${TEST_HEADER}")
    file(APPEND "${GENERATED_TESTS}" "#include <${RELATIVE_HEADER}>\n")
endforeach()

add_executable(sqlite_mkkeywordhash_tests main.cpp "${GENERATED_TESTS}")
target_include_directories(sqlite_mkkeywordhash_tests PRIVATE "${SQLITE_MKKEYWORDHASH_TEST_INCLUDE_ROOT}")
target_link_libraries(sqlite_mkkeywordhash_tests PRIVATE sqlite::utils)
target_compile_features(sqlite_mkkeywordhash_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(sqlite_mkkeywordhash_tests PRIVATE /W4 /permissive-)
else()
    target_compile_options(sqlite_mkkeywordhash_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_test(NAME sqlite-mkkeywordhash COMMAND sqlite_mkkeywordhash_tests)
