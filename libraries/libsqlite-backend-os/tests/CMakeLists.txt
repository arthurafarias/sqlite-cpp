# SRS FR-11/FR-12/FR-6: every testing/*_test.hpp header under this library's
# include/ tree self-registers a test_group on inclusion, mirroring
# sqlite-utils's tests/CMakeLists.txt. FR-6's "two implementations selected
# by CMake if()" adapts, for a header-only library, to selecting *which
# platform's testing headers get included into this generated translation
# unit* -- unix/testing/*_test.hpp exercise real POSIX file I/O and can only
# run where POSIX exists; windows/testing/*_test.hpp exercise real Win32 file
# I/O and can only run where the Win32 API (or a sufficiently complete
# emulation of it) exists. On a non-Windows host, verify the windows/ side by
# cross-compiling with cmake/mingw-w64-toolchain.cmake and running the
# resulting .exe under Wine -- see docs/index.md's "Verifying the Windows
# body from Linux" section for the exact commands; this is a documented
# manual workflow, not wired into the default ctest run, since it depends on
# a mingw-w64 + Wine host setup this library does not otherwise require.
set(SQLITE_BACKEND_OS_TEST_INCLUDE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../include")

set(SQLITE_BACKEND_OS_TEST_HEADERS)
file(GLOB_RECURSE SQLITE_BACKEND_OS_SHARED_TEST_HEADERS CONFIGURE_DEPENDS
    "${SQLITE_BACKEND_OS_TEST_INCLUDE_ROOT}/sqlite/backend/os/testing/*_test.hpp")
list(APPEND SQLITE_BACKEND_OS_TEST_HEADERS ${SQLITE_BACKEND_OS_SHARED_TEST_HEADERS})

if(WIN32)
    file(GLOB_RECURSE SQLITE_BACKEND_OS_PLATFORM_TEST_HEADERS CONFIGURE_DEPENDS
        "${SQLITE_BACKEND_OS_TEST_INCLUDE_ROOT}/sqlite/backend/os/windows/testing/*_test.hpp")
else()
    file(GLOB_RECURSE SQLITE_BACKEND_OS_PLATFORM_TEST_HEADERS CONFIGURE_DEPENDS
        "${SQLITE_BACKEND_OS_TEST_INCLUDE_ROOT}/sqlite/backend/os/unix/testing/*_test.hpp")
endif()
list(APPEND SQLITE_BACKEND_OS_TEST_HEADERS ${SQLITE_BACKEND_OS_PLATFORM_TEST_HEADERS})

set(GENERATED_TESTS "${CMAKE_CURRENT_BINARY_DIR}/all_tests.cpp")
file(WRITE "${GENERATED_TESTS}" "// Generated by CMake. Every testing/*_test.hpp header self-registers a test_group.\n")
foreach(TEST_HEADER IN LISTS SQLITE_BACKEND_OS_TEST_HEADERS)
    file(RELATIVE_PATH RELATIVE_HEADER "${SQLITE_BACKEND_OS_TEST_INCLUDE_ROOT}" "${TEST_HEADER}")
    file(APPEND "${GENERATED_TESTS}" "#include <${RELATIVE_HEADER}>\n")
endforeach()

add_executable(sqlite_backend_os_tests main.cpp "${GENERATED_TESTS}")
target_link_libraries(sqlite_backend_os_tests PRIVATE sqlite::backend::os)
target_compile_features(sqlite_backend_os_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(sqlite_backend_os_tests PRIVATE /W4 /permissive-)
else()
    target_compile_options(sqlite_backend_os_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_test(NAME sqlite-backend-os COMMAND sqlite_backend_os_tests)
