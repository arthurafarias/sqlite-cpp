# SRS FR-11/FR-12: mirrors the other sqlite-cpp libraries'
# tests/CMakeLists.txt -- every testing/*_test.hpp header under this
# library's include/ tree self-registers a test_group on inclusion.
set(SQLITE_COMMAND_PROCESSOR_TEST_INCLUDE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../include")

set(GENERATED_TESTS "${CMAKE_CURRENT_BINARY_DIR}/all_tests.cpp")
file(WRITE "${GENERATED_TESTS}" "// Generated by CMake. Every testing/*_test.hpp header self-registers a test_group.\n")
file(GLOB_RECURSE SQLITE_COMMAND_PROCESSOR_TEST_HEADERS CONFIGURE_DEPENDS "${SQLITE_COMMAND_PROCESSOR_TEST_INCLUDE_ROOT}/*_test.hpp")
foreach(TEST_HEADER IN LISTS SQLITE_COMMAND_PROCESSOR_TEST_HEADERS)
    file(RELATIVE_PATH RELATIVE_HEADER "${SQLITE_COMMAND_PROCESSOR_TEST_INCLUDE_ROOT}" "${TEST_HEADER}")
    file(APPEND "${GENERATED_TESTS}" "#include <${RELATIVE_HEADER}>\n")
endforeach()

add_executable(sqlite_core_command_processor_tests main.cpp "${GENERATED_TESTS}")
target_link_libraries(sqlite_core_command_processor_tests PRIVATE sqlite::core::command_processor)
target_compile_features(sqlite_core_command_processor_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(sqlite_core_command_processor_tests PRIVATE /W4 /permissive-)
else()
    target_compile_options(sqlite_core_command_processor_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_test(NAME sqlite-core-command-processor COMMAND sqlite_core_command_processor_tests)
