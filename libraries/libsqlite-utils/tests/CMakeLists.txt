# SRS FR-11/FR-12: every testing/*_test.hpp header under this library's
# include/ tree self-registers a test_group when included; this glob-and-
# generate step (mirroring opheap's tests/CMakeLists.txt) collects them all
# into one generated translation unit so the library's test binary is
# runnable both standalone (`ctest -R sqlite-utils`) and as part of a
# workspace-wide `ctest` run (FR-12).
set(SQLITE_UTILS_TEST_INCLUDE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../include")

set(GENERATED_TESTS "${CMAKE_CURRENT_BINARY_DIR}/all_tests.cpp")
file(WRITE "${GENERATED_TESTS}" "// Generated by CMake. Every testing/*_test.hpp header self-registers a test_group.\n")
file(GLOB_RECURSE SQLITE_UTILS_TEST_HEADERS CONFIGURE_DEPENDS "${SQLITE_UTILS_TEST_INCLUDE_ROOT}/*_test.hpp")
foreach(TEST_HEADER IN LISTS SQLITE_UTILS_TEST_HEADERS)
    file(RELATIVE_PATH RELATIVE_HEADER "${SQLITE_UTILS_TEST_INCLUDE_ROOT}" "${TEST_HEADER}")
    file(APPEND "${GENERATED_TESTS}" "#include <${RELATIVE_HEADER}>\n")
endforeach()

add_executable(sqlite_utils_tests main.cpp "${GENERATED_TESTS}")
target_link_libraries(sqlite_utils_tests PRIVATE sqlite::utils)
target_compile_features(sqlite_utils_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(sqlite_utils_tests PRIVATE /W4 /permissive-)
else()
    target_compile_options(sqlite_utils_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

# Named "sqlite-utils" (not "sqlite_utils_tests") so `ctest -R sqlite-utils`
# runs exactly this library's test binary, per FR-12.
add_test(NAME sqlite-utils COMMAND sqlite_utils_tests)
