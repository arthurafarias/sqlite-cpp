# SRS FR-11/FR-12: mirrors the other sqlite-cpp libraries'
# tests/CMakeLists.txt -- every testing/*_test.hpp header under this
# library's include/ tree self-registers a test_group on inclusion.
#
# Unlike sqlite-backend-os, this library's tests are not split by platform:
# page_store.hpp (this library's temporary pager stand-in, see docs/index.md)
# currently hardcodes sqlite::backend::os::unix directly rather than
# selecting unix/windows structurally, so these tests only run natively on
# POSIX hosts for now.
set(SQLITE_TREE_TEST_INCLUDE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../include")

set(GENERATED_TESTS "${CMAKE_CURRENT_BINARY_DIR}/all_tests.cpp")
file(WRITE "${GENERATED_TESTS}" "// Generated by CMake. Every testing/*_test.hpp header self-registers a test_group.\n")
file(GLOB_RECURSE SQLITE_TREE_TEST_HEADERS CONFIGURE_DEPENDS "${SQLITE_TREE_TEST_INCLUDE_ROOT}/*_test.hpp")
foreach(TEST_HEADER IN LISTS SQLITE_TREE_TEST_HEADERS)
    file(RELATIVE_PATH RELATIVE_HEADER "${SQLITE_TREE_TEST_INCLUDE_ROOT}" "${TEST_HEADER}")
    file(APPEND "${GENERATED_TESTS}" "#include <${RELATIVE_HEADER}>\n")
endforeach()

add_executable(sqlite_backend_tree_tests main.cpp "${GENERATED_TESTS}")
target_link_libraries(sqlite_backend_tree_tests PRIVATE sqlite::backend::tree)
target_compile_features(sqlite_backend_tree_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(sqlite_backend_tree_tests PRIVATE /W4 /permissive-)
else()
    target_compile_options(sqlite_backend_tree_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_test(NAME sqlite-backend-tree COMMAND sqlite_backend_tree_tests)
